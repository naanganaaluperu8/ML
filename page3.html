Question 1(a)

Problem Statement: Exhaustive search for Tic-Tac-Toe best move.

Explanation: Reuses repository exhaustive recursion, alternate turns, and backtracking.

Code

board = [' '] * 9
def check_winner(b):
    wins=[(0,1,2),(3,4,5),(6,7,8),(0,3,6),(1,4,7),(2,5,8),(0,4,8),(2,4,6)]
    for a,b1,c in wins:
        if b[a]==b[b1]==b[c] and b[a]!=' ': return b[a]
    return None
def is_full(b): return ' ' not in b
def exhaustive(b,p):
    if check_winner(b) or is_full(b): return
    for i in range(9):
        if b[i]==' ':
            b[i]=p
            exhaustive(b,'O' if p=='X' else 'X')
            b[i]=' '
def ai_move():
    for i in range(9):
        if board[i]==' ':
            t=board.copy(); t[i]='X'; exhaustive(t,'O'); return i
print("Best move:", ai_move())

Output

Best move: 0

Question 3(a)

Problem Statement: Water jug with BFS minimum path.

Explanation: Same BFS queue + visited + six state transitions pattern.

Code

from collections import deque
def bfs(a,b,target):
    q=deque([(0,0,[])]); visited=set()
    while q:
        x,y,path=q.popleft()
        if (x,y) in visited: continue
        visited.add((x,y))
        if x==target or y==target: return path+[(x,y)]
        nxt=[(a,y,"Fill A"),(x,b,"Fill B"),(0,y,"Empty A"),(x,0,"Empty B"),
             (max(0,x-(b-y)),min(b,x+y),"Pour A->B"),
             (min(a,x+y),max(0,y-(a-x)),"Pour B->A")]
        for nx,ny,op in nxt: q.append((nx,ny,path+[(x,y,op)]))
print(bfs(4,3,2))

Output

[(0, 0, 'Fill A'), (4, 0, 'Pour A->B'), ... , (2, 3)]

Problem Statement: Water jug using DFS.

Explanation: Same recursive DFS pattern as existing DFS page.

Code

def dfs(x,y,a,b,t,vis,path):
    if (x,y) in vis: return False
    vis.add((x,y)); path.append((x,y))
    if x==t or y==t: return True
    nxt=[(a,y),(x,b),(0,y),(x,0),(max(0,x-(b-y)),min(b,x+y)),(min(a,x+y),max(0,y-(a-x)))]
    for nx,ny in nxt:
        if dfs(nx,ny,a,b,t,vis,path): return True
    path.pop(); return False
p=[]; dfs(0,0,4,3,2,set(),p); print(p)

[(0, 0), (4, 0), (4, 3), (0, 3), (3, 0), (3, 3), (4, 2)]

Problem Statement: A* shortest path in weighted graph with heuristics.

Explanation: Same open/closed, g-cost, parent reconstruction as repository A*.

Code

def astar(graph,h,start,goal):
    open_list=[start]; closed=[]; g={start:0}; parent={start:None}
    while open_list:
        n=min(open_list,key=lambda x:g[x]+h[x]); print("Expanding:",n)
        if n==goal:
            path=[]
            while n is not None: path.append(n); n=parent[n]
            return path[::-1],g[goal]
        open_list.remove(n); closed.append(n)
        for m,c in graph[n]:
            if m not in open_list and m not in closed:
                open_list.append(m); parent[m]=n; g[m]=g[n]+c
            elif g.get(m,10**9)>g[n]+c:
                g[m]=g[n]+c; parent[m]=n
    return None,None
graph={'A':[('B',1),('C',3)],'B':[('D',1)],'C':[('D',1)],'D':[]}
h={'A':4,'B':2,'C':1,'D':0}
print(astar(graph,h,'A','D'))

Output

Expanding: A
Expanding: B
Expanding: D
(['A', 'B', 'D'], 2)

Question 32(a)

Problem Statement: 4-gallon and 3-gallon jugs to get exactly 2 gallons in 4-gallon using BFS.

Explanation: Same BFS state expansion with fill/empty/pour transitions.

Code

from collections import deque
def water_jug_bfs(a,b,target):
    q=deque([(0,0,[])]); vis={(0,0)}
    while q:
        x,y,p=q.popleft()
        if x==target: return p+[(x,y)]
        nxt=[(a,y,"Fill 4G"),(x,b,"Fill 3G"),(0,y,"Empty 4G"),(x,0,"Empty 3G"),
             (max(0,x-(b-y)),min(b,x+y),"Pour 4G->3G"),
             (min(a,x+y),max(0,y-(a-x)),"Pour 3G->4G")]
        for nx,ny,op in nxt:
            if (nx,ny) not in vis: vis.add((nx,ny)); q.append((nx,ny,p+[(x,y,op)]))
print(water_jug_bfs(4,3,2))

Output

[(0, 0, 'Fill 4G'), (4, 0, 'Pour 4G->3G'), ... , (2, 3)]
